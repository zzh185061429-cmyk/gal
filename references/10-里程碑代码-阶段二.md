```ts
  '示例咖啡店':            'https://i.postimg.cc/xxxx/bg-cafe.png',
  '示例咖啡店/靠窗座位':    'https://i.postimg.cc/xxxx/bg-cafe-window.png',

};

export function spriteOf(name: string, emotion?: string): string | undefined {
  const c = CHARACTERS.find(k => k.name === name);
  if (!c) return undefined;
  const key = EMOTION_MAP[emotion ?? '默认'] ?? EMOTION_MAP['默认'];
  return c.sprites[key] ?? c.sprites[EMOTION_MAP['默认']];   // 缺差分回退默认（踩坑 7）
}

export function sceneOf(path?: string): string | undefined {
  if (!path) return undefined;
  if (SCENES[path]) return SCENES[path];
  const parts = path.split('/');
  while (parts.length > 1) {
    parts.pop();
    const up = parts.join('/');
    if (SCENES[up]) return SCENES[up];                       // 逐级回退（踩坑 13）
  }
  return undefined;
}
```

## src/parser.ts（在阶段一基础上替换——新增 location 游标 + sprite 解析）

```ts
import { spriteOf } from './assets';

export interface ScriptLine {
  type: 'narrator' | 'dialog' | 'thought';
  speaker?: string;
  emotion?: string;
  text: string;
  /** 由 [scene:…] 标签继承，渲染背景用 */
  location?: { path: string; displayName: string };
  /** 解析期就定好的立绘 URL（<user> 恒为 undefined——不占立绘位，踩坑 9） */
  sprite?: string;
}

const SCENE_RE = /^\[scene:([^\]]+?)\]$/;

export function parseScriptContent(rawText: string, playerName?: string): ScriptLine[] {
  const cleaned = stripThinking(rawText);
  const m = cleaned.match(/<content>([\s\S]*?)<\/content>/);
  if (!m) return [];
  const lines: ScriptLine[] = [];
  let currentLocation: { path: string; displayName: string } | undefined;   // 场景游标

  for (const seg of m[1].split(/\n/)) {
    const line = seg.trim();
    if (!line) continue;

    /* 场景标签：不生成卡片，只挪游标（踩坑：首楼 <content> 第一行就该是它） */
    const sm = line.match(SCENE_RE);
    if (sm) {
      const path = sm[1].trim();
      const parts = path.split('/').map(s => s.trim()).filter(Boolean);
      currentLocation = { path, displayName: parts[parts.length - 1] ?? path };
      continue;
    }

    let mt: RegExpMatchArray | null;
    if ((mt = line.match(DIALOG_RE))) {
      const speaker = mt[1].trim();
      const isU = isUser(speaker, playerName);
      lines.push({
        type: 'dialog',
        speaker: isU ? '<user>' : speaker,
        emotion: isU ? undefined : mt[2].trim(),
        text: mt[3],
        location: currentLocation,
        sprite: isU ? undefined : spriteOf(speaker, mt[2].trim()),
      });
    } else if ((mt = line.match(USER_DLG))) {
      lines.push({ type: 'dialog', speaker: '<user>', text: mt[1], location: currentLocation });
    } else if ((mt = line.match(THOUGHT_RE))) {
      const speaker = mt[1].trim();
      const isU = isUser(speaker, playerName);
      lines.push({
        type: 'thought',
        speaker: isU ? '<user>' : speaker,
        emotion: isU ? undefined : mt[2].trim(),
        text: mt[3],
        location: currentLocation,
        sprite: isU ? undefined : spriteOf(speaker, mt[2].trim()),
      });
    } else if ((mt = line.match(USER_THO))) {
      lines.push({ type: 'thought', speaker: '<user>', text: mt[1], location: currentLocation });
    } else {
      lines.push({ type: 'narrator', text: line, location: currentLocation });
    }
  }
  return lines;
}
```

## src/App.tsx（替换——新增背景层、立绘舞台、预加载）

```tsx
import React, { useCallback, useEffect, useRef, useState } from 'react';
import { parseScriptContent, isUser, type ScriptLine } from './parser';
import { TypingText } from './TypingText';
import { sceneOf } from './assets';
import { SAMPLE_FLOOR, SPEED_MS } from './fixture';   // 把阶段一的夹具与速度常量挪到这里

interface StageCharacter {
  speaker: string; sprite: string;
  position: 'left' | 'center' | 'right';
  isActive: boolean; hidden: boolean;
}

export default function App() {
  const [script] = useState<ScriptLine[]>(() => parseScriptContent(SAMPLE_FLOOR));
  const [index, setIndex] = useState(0);
  const [isTyping, setIsTyping] = useState(false);
  const skipRef = useRef(false);
  const line = script[index];

  /* ── 立绘舞台装配（照抄 幻璃镜 GameScreen.tsx:544-575 的状态机）── */
  const [cast, setCast] = useState<StageCharacter[]>([]);
  const prevLocRef = useRef<string | null>(null);

  useEffect(() => {
    const loc = line?.location?.path ?? null;
    const locChanged = loc !== prevLocRef.current;
    prevLocRef.current = loc;

    // 旁白 / 玩家发言：全员暂退（保留位置记忆），换景则清台（踩坑 11）
    if (!line?.speaker || line.type === 'narrator' || isUser(line.speaker)) {
      if (locChanged) setCast([]);
      else setCast(prev => prev.map(c => ({ ...c, isActive: false, hidden: true })));
      return;
    }
    const sprite = line.sprite ?? '';
    if (locChanged) {
      setCast([{ speaker: line.speaker, sprite, position: 'center', isActive: true, hidden: false }]);
      return;
    }
    setCast(prev => {
      const idx = prev.findIndex(c => c.speaker === line.speaker);
      if (idx >= 0) {
        return prev.map((c, i) => i === idx
          ? { ...c, sprite: sprite || c.sprite, isActive: true, hidden: false }
          : { ...c, isActive: false, hidden: false });
      }
      // 新角色：center → right → left，slice(-3) 硬性同屏 ≤3（踩坑 10）
      const position: StageCharacter['position'] =
        prev.length === 0 ? 'center' : prev.length === 1 ? 'right' : 'left';
      const kept = [...prev.filter(c => c.position !== position),
        { speaker: line.speaker!, sprite, position, isActive: true, hidden: false }].slice(-3);
      return kept.map((c, i) => ({ ...c, isActive: i === kept.length - 1 }));
    });
  }, [line]);

  /* ── 立绘预加载：解析完预热全部差分（踩坑 12）── */
  useEffect(() => {
    for (const l of script) {
      if (l.sprite) { const img = new Image(); img.src = l.sprite; }
    }
  }, [script]);

  const next = useCallback(() => {
    if (!line) return;
    if (isTyping) { skipRef.current = true; return; }
    if (index < script.length - 1) setIndex(i => i + 1);
  }, [line, isTyping, index, script.length]);
  const prev = useCallback(() => { if (index > 0) setIndex(i => i - 1); }, [index]);

  useEffect(() => { /* 键盘映射与阶段一相同，此处省略 */ }, [next, prev]);

  const bg = sceneOf(line?.location?.path);
  const bgKey = line?.location?.path ?? 'none';

  if (!line) {
    return <div className="gal-root"><div className="gal-empty">本楼无 &lt;content&gt; 正文</div></div>;
  }

  return (
    <div className="gal-root" onClick={next}>
      {/* 背景层：location 变化才换图；查不到保留旧背景不闪黑 */}
      {bg && <img key={bgKey} className="gal-bg" src={bg} alt="" />}

      {/* 立绘层：说话人亮色，旁听暗色；hidden 暂退 */}
      <div className="gal-stage">
        {cast.map(c => (
          <div key={c.speaker}
            className={'gal-sprite pos-' + c.position + (c.hidden ? ' is-hidden' : '')}
            style={{ filter: c.isActive ? 'none' : 'brightness(0.55) grayscale(0.35)' }}>
            {c.sprite && <img src={c.sprite} alt={c.speaker} />}
          </div>
        ))}
      </div>

      <div className="gal-textbox">
        {line.speaker && (
          <div className={'gal-name' + (line.speaker === '<user>' ? ' gal-name-user' : '')}>
            {line.speaker === '<user>' ? '我' : line.speaker}
          </div>
        )}
        <TypingText key={index} text={line.text} speedMs={SPEED_MS} lineType={line.type}
          onTypingStateChange={setIsTyping} skipRef={skipRef} />
        <div className="gal-progress">{index + 1} / {script.length}</div>
      </div>
    </div>
  );
}
```
