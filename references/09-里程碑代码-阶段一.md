## src/parser.ts（全文）

```ts
export type LineType = 'narrator' | 'dialog' | 'thought';

export interface ScriptLine {
  type: LineType;
  speaker?: string;   // '<user>' 或角色名
  emotion?: string;   // 差分标签，如 '生气'
  text: string;
}

/* ── 思维链剥离：必须在提取 <content> 之前执行（踩坑 4/5）── */
const THINKING_PAIRS: [string, string][] = [
  ['<think>', '</think>'],
  ['<thinking>', '</thinking>'],
  ['<Chain_of_Thought>', '</Chain_of_Thought>'],
  ['<draft>', '</draft>'],
  ['<simple_thinking>', '</simple_thinking>'],
];

function escapeRe(s: string): string {
  return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

export function stripThinking(raw: string): string {
  let text = raw;
  for (const [open, close] of THINKING_PAIRS) {
    text = text.replace(new RegExp(escapeRe(open) + '[\\s\\S]*?' + escapeRe(close), 'gi'), '');
  }
  text = text.replace(/[\s\S]*?<\/konatan_planning~>/i, ''); // 前缀截断
  return text;
}

/* ── 行协议：全行锚定，半角引号冒号 ── */
const DIALOG_RE  = /^(.+?)\[(.+?)\]:"(.+)"$/s;   // 角色名[情绪]:"台词"
const USER_DLG   = /^<user>:"(.+)"$/s;           // <user>:"台词"
const THOUGHT_RE = /^(.+?)\[(.+?)\]:\*(.+)\*$/s; // 角色名[情绪]:*独白*
const USER_THO   = /^<user>:\*(.+)\*$/s;         // <user>:*独白*

/** <user> / 我 / 玩家自定义名 都算玩家（踩坑 6） */
export function isUser(speaker: string, playerName?: string): boolean {
  return speaker === '<user>' || speaker === '我' || (!!playerName && speaker === playerName);
}

export function parseScriptContent(rawText: string, playerName?: string): ScriptLine[] {
  const cleaned = stripThinking(rawText);                    // ① 先剥思维链
  const m = cleaned.match(/<content>([\s\S]*?)<\/content>/); // ② 再提取正文
  if (!m) return [];
  const lines: ScriptLine[] = [];
  for (const seg of m[1].split(/\n/)) {
    const line = seg.trim();
    if (!line) continue;
    let mt: RegExpMatchArray | null;
    if ((mt = line.match(DIALOG_RE))) {
      const speaker = mt[1].trim();
      if (isUser(speaker, playerName)) {
        // AI 写了 <user>[默认]:"…" 之类 → 按玩家行处理，无差分
        lines.push({ type: 'dialog', speaker: '<user>', text: mt[3] });
      } else {
        lines.push({ type: 'dialog', speaker, emotion: mt[2].trim(), text: mt[3] });
      }
    } else if ((mt = line.match(USER_DLG))) {
      lines.push({ type: 'dialog', speaker: '<user>', text: mt[1] });
    } else if ((mt = line.match(THOUGHT_RE))) {
      const speaker = mt[1].trim();
      lines.push(isUser(speaker, playerName)
        ? { type: 'thought', speaker: '<user>', text: mt[3] }
        : { type: 'thought', speaker, emotion: mt[2].trim(), text: mt[3] });
    } else if ((mt = line.match(USER_THO))) {
      lines.push({ type: 'thought', speaker: '<user>', text: mt[1] });
    } else {
      lines.push({ type: 'narrator', text: line });          // ③ 兜底 = 旁白
    }
  }
  return lines;
}
```

## src/TypingText.tsx（全文）

> 红线组件：打字状态**内部自持**，父组件只发全文、收"打字中/完毕"回调。父组件必须传**稳定引用**的回调（如 `setIsTyping`），memo 比较器才能把无关重渲挡在门外。

```tsx
import React, { useCallback, useEffect, useState } from 'react';
import type { LineType } from './parser';

interface TypingTextProps {
  text: string;              // 整行全文
  speedMs: number;           // 每字毫秒；0 = 瞬发
  lineType: LineType;
  onTypingStateChange?: (typing: boolean) => void;
  skipRef: React.MutableRefObject<boolean>; // 置 true = 立即满行
}

export const TypingText = React.memo(function TypingText({
  text, speedMs, lineType, onTypingStateChange, skipRef,
}: TypingTextProps) {
  const [shown, setShown] = useState('');
  const [typing, setTyping] = useState(false);

  const notify = useCallback((t: boolean) => {
    setTyping(t);
    onTypingStateChange?.(t);
  }, [onTypingStateChange]);

  useEffect(() => {
    let raf = 0;
    let cancelled = false;
    skipRef.current = false;

    if (speedMs <= 0) { setShown(text); notify(false); return; }
    setShown(''); notify(true);

    let i = 0;
    let last = performance.now();
    const tick = (now: number) => {
      if (cancelled) return;
      if (skipRef.current) { setShown(text); notify(false); return; }
      if (now - last < speedMs) { raf = requestAnimationFrame(tick); return; }
      last = now;
      if (i < text.length) {
        i += 1;
        setShown(text.slice(0, i));
        raf = requestAnimationFrame(tick);
      } else {
        notify(false);
      }
    };
    raf = requestAnimationFrame(tick);
    return () => { cancelled = true; cancelAnimationFrame(raf); };
  }, [text, speedMs, skipRef, notify]);

  return (
    <div className={'gal-text ' + lineType}>
      {shown}
      {typing && <span className="gal-cursor">▍</span>}
    </div>
  );
}, (p, n) =>
  p.text === n.text && p.speedMs === n.speedMs &&
  p.lineType === n.lineType && p.isSkipping === n.isSkipping);
```

> 注意比较器里出现 `isSkipping` 但本简化版没传——要支持按住 Ctrl 连跳，就加回该 prop（参照 `幻璃镜 TypingText.tsx:59-64`）。

## src/App.tsx（全文）

```tsx
import React, { useCallback, useEffect, useRef, useState } from 'react';
import { parseScriptContent, type ScriptLine } from './parser';
import { TypingText } from './TypingText';

/** 测试夹具：浏览器裸跑用。接入酒馆后换成 getChatMessages(楼层号)[0].message */
const SAMPLE_FLOOR = `
<content>
午后三点的阳光斜进店里，灰尘在光柱里慢慢浮沉。

小棠[开心]:"你可算来了，我再等十分钟就要把你那份蛋糕吃掉了。"

她把菜单推到桌子对面，指尖在桌面上敲了敲。

<user>:"路上堵车，罚我下午茶请客。"

小棠[害羞]:*其实……我点两份也吃得下。*

电视里传来"特价商品9块9包邮"的声音，没有人理会它。

小棠[生气]:"哼，那你昨天还排队买奶茶，说好的减肥呢？"
</content>
`;

const SPEED_MS = 35; // 每字毫秒；0 = 瞬发

export default function App() {
  const [script] = useState<ScriptLine[]>(() => parseScriptContent(SAMPLE_FLOOR));
  const [index, setIndex] = useState(0);
  const [isTyping, setIsTyping] = useState(false);
  const skipRef = useRef(false);

  const line = script[index];

  const next = useCallback(() => {
    if (!line) return;
    if (isTyping) { skipRef.current = true; return; }        // 打字中第一次点击 = 跳过
    if (index < script.length - 1) setIndex(i => i + 1);
    // 本楼到头 → 下一楼：阶段三在这里接入（SKILL.md §S4）
  }, [line, isTyping, index, script.length]);

  const prev = useCallback(() => {
    if (index > 0) setIndex(i => i - 1);
  }, [index]);

  // 键盘：Enter/Space/→ 前进，← 回退（输入框聚焦时不抢）
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      const t = e.target as HTMLElement;
      if (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA') return;
      if (e.key === 'Enter' || e.key === ' ' || e.key === 'ArrowRight') { e.preventDefault(); next(); }
      else if (e.key === 'ArrowLeft') { e.preventDefault(); prev(); }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [next, prev]);

  if (!line) {
    return <div className="gal-root"><div className="gal-empty">本楼无 &lt;content&gt; 正文（检查 AI 输出格式）</div></div>;
  }

  return (
    <div className="gal-root" onClick={next}>
      {/* 背景层：阶段二 */}
      {/* 立绘层：阶段二 */}
      <div className="gal-textbox">
        {line.speaker && (
          <div className={'gal-name' + (line.speaker === '<user>' ? ' gal-name-user' : '')}>
            {line.speaker === '<user>' ? '我' : line.speaker}
          </div>
        )}
        <TypingText
          key={index}                 // 换行重挂载，打字机状态归零
          text={line.text}
          speedMs={SPEED_MS}
          lineType={line.type}
          onTypingStateChange={setIsTyping}
          skipRef={skipRef}
        />
        <div className="gal-progress">{index + 1} / {script.length}</div>
      </div>
    </div>
  );
}
```
