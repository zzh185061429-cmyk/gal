```ts
/* ═══ 酒馆全局函数安全层 ═══
 * 浏览器裸跑时这些全局不存在 → 全部判空降级；真机 iframe 内原样可用。
 * 项目纪律：除本文件外，任何地方不得出现裸的 getChatMessages/triggerSlash… */

const g = globalThis as any;

/** 是否运行在酒馆 iframe 内 */
export const hasTavern = typeof g.getChatMessages === 'function';

/** 读楼层消息。range 支持楼层号、'a-b'、'0-{{lastMessageId}}'；可带 { role } 过滤 */
export function getMessages(range: number | string, opts?: any): any[] {
  try { return g.getChatMessages(range, opts) ?? []; } catch { return []; }
}

/** 最后一条消息的楼层号；无酒馆/空聊天返回 null */
export function lastMessageId(): number | null {
  try { return g.getLastMessageId() ?? null; } catch { return null; }
}

/** 执行 slash 命令；无酒馆环境返回 false（调用方据此降级） */
export async function slash(cmd: string): Promise<boolean> {
  if (typeof g.triggerSlash !== 'function') return false;
  try { await g.triggerSlash(cmd); return true; }
  catch (e) { console.error('[gal] slash 失败:', cmd, e); return false; }
}

/** 订阅事件，返回退订函数 */
export function onEvent(event: string | undefined, cb: (...args: any[]) => void): () => void {
  if (!event || typeof g.eventOn !== 'function') return () => {};
  try {
    const ret = g.eventOn(event, cb);
    return () => { try { ret?.stop?.(); } catch { /* noop */ } };
  } catch { return () => {}; }
}

/** 发自定义事件（13 章命名空间规范：事件名必须带项目前缀） */
export function emit(event: string, ...args: any[]): void {
  try { g.eventEmit?.(event, ...args); } catch { /* noop */ }
}

/** 聊天变量读/写 */
export function chatVars(): any {
  try { return g.getVariables?.({ type: 'chat' }) ?? {}; } catch { return {}; }
}
export function updateChatVars(fn: (vars: any) => any): void {
  try { g.updateVariablesWith?.(fn, { type: 'chat' }); } catch { /* noop */ }
}

/** 酒馆/iframe 事件名集合（环境缺失时为空对象 → onEvent 收到 undefined 自动跳过） */
export const TE: any = (typeof g.tavern_events === 'object' && g.tavern_events) || {};
export const IE: any = (typeof g.iframe_events === 'object' && g.iframe_events) || {};

/** 本项目自定义刷新事件（删楼/重roll 完成后通知各楼 iframe 重解析） */
export const STORY_UPDATED = 'galmilestone_story_updated';
```

## src/floors.ts（全文）

```ts
import { getMessages, lastMessageId } from './tavern';

/** 扫描所有 AI 楼层号（源码：两项目 floorNav.ts 一字不差） */
export function getAssistantFloors(): number[] {
  const msgs = getMessages('0-{{lastMessageId}}', { role: 'assistant' });
  return msgs.map((m: any) => m.message_id).sort((a: number, b: number) => a - b);
}

/** 最新 AI 楼层号——最后楼可能是 user，必须校验角色（SKILL.md §S4） */
export function getLatestAssistantId(): number | null {
  const lastId = lastMessageId();
  if (lastId == null) return null;
  const msg = getMessages(lastId)[0];
  if (msg?.role === 'assistant') return msg.message_id;
  if (lastId > 0) {
    const prev = getMessages(lastId - 1)[0];
    if (prev?.role === 'assistant') return prev.message_id;
  }
  return null;
}

/** 读某楼正文原文 */
export function floorMessage(floorId: number | null): string | null {
  if (floorId == null) return null;
  return getMessages(floorId)[0]?.message ?? null;
}
```

## src/App.tsx（替换——阶段二全部功能 + 本阶段新增，全文）

```tsx
import React, { useCallback, useEffect, useMemo, useRef, useState } from 'react';
import { parseScriptContent, isUser, type ScriptLine } from './parser';
import { TypingText } from './TypingText';
import { sceneOf } from './assets';
import { SAMPLE_FLOOR, SPEED_MS } from './fixture';
import { hasTavern, getMessages, slash, onEvent, TE, IE } from './tavern';
import { getAssistantFloors, getLatestAssistantId, floorMessage } from './floors';

interface StageCharacter {
  speaker: string; sprite: string;
  position: 'left' | 'center' | 'right';
  isActive: boolean; hidden: boolean;
}

export default function App() {
  /* ── 楼层状态（SKILL.md §S4 三态机）── */
  const [viewingFloorId, setViewingFloorId] = useState<number | null>(null); // null=跟随最新
  const [lastAssistantFloorId, setLastAssistantFloorId] = useState<number | null>(null);
  const [generatingFloorId, setGeneratingFloorId] = useState<number | null>(null);
  const [isGenerating, setIsGenerating] = useState(false);
  const [floors, setFloors] = useState<number[]>([]);
  const isGenRef = useRef(false); isGenRef.current = isGenerating;

  /* ── 播放状态（阶段一/二）── */
  const [index, setIndex] = useState(0);
  const [isTyping, setIsTyping] = useState(false);
  const skipRef = useRef(false);
  const progressRef = useRef(new Map<number, number>());            // 楼号→卡号
  const lastParseRef = useRef<{ floor: number | null; content: string }>({ floor: null, content: '' });

  /* ── 消息 → 解析（浏览器裸跑回落夹具）── */
  const targetFloorId = viewingFloorId ?? lastAssistantFloorId;
  const rawMessage = hasTavern ? floorMessage(targetFloorId) : SAMPLE_FLOOR;
  const script = useMemo(() => parseScriptContent(rawMessage ?? ''), [rawMessage]);
  const line = script[index];

  /* ── 楼层数据同步（MESSAGE_RECEIVED/UPDATED/GENERATION_ENDED/CHAT_CHANGED）──
   * 简化说明：流式期间的防误重置靠"内容变了才清进度"兜底（下两段 effect），
   * 正式项目再按 SKILL.md §S4 防抖①补 messageId 比对。 */
  useEffect(() => {
    if (!hasTavern) return;
    const sync = () => {
      const latestId = getLatestAssistantId();
      if (latestId != null) {
        if (isGenRef.current) setGeneratingFloorId(p => (p != null && latestId <= p ? p : latestId));
        else setLastAssistantFloorId(latestId);
      }
      setFloors(getAssistantFloors());
    };
    sync();
    const offs = [
      onEvent(TE.MESSAGE_RECEIVED, sync),
      onEvent(TE.MESSAGE_UPDATED, sync),
      onEvent(IE.GENERATION_ENDED, () => setTimeout(sync, 300)),
      onEvent(TE.CHAT_CHANGED, () => { setViewingFloorId(null); sync(); }),
    ];
    return () => offs.forEach(off => off());
  }, []);

  /* ── 生成锁：钉画面 → 解锁自动跳最新（SKILL.md §S4 生成锁，成对调用！）── */
  const startGenerating = useCallback(() => {
    setViewingFloorId(viewingFloorId ?? lastAssistantFloorId);   // 钉死当前画面楼
    setIsGenerating(true);
  }, [viewingFloorId, lastAssistantFloorId]);

  const finishGenerating = useCallback(() => {
    setIsGenerating(false);
    const latest = getLatestAssistantId();
    if (latest != null) { setGeneratingFloorId(latest); setLastAssistantFloorId(latest); }
    setViewingFloorId(null);   // ★ 解锁 = 归位 null = 自动跳最新楼
  }, []);

  /* ── 阅读进度：换楼恢复；仅本楼内容真变（重生成）才清零（踩坑 17）── */
  useEffect(() => {
    if (targetFloorId == null) return;
    const prev = lastParseRef.current;
    if (prev.floor === targetFloorId && prev.content !== (rawMessage ?? '')) {
      progressRef.current.delete(targetFloorId);
    }
    lastParseRef.current = { floor: targetFloorId, content: rawMessage ?? '' };
    const saved = progressRef.current.get(targetFloorId);
    setIndex(saved != null && saved < script.length ? saved : 0);
  }, [targetFloorId, rawMessage, script.length]);

  useEffect(() => {
    if (targetFloorId != null) progressRef.current.set(targetFloorId, index);
  }, [targetFloorId, index]);

  /* ── 楼层导航（生成中只许翻到生成楼之前——SKILL.md §S4）── */
  const availableFloors = useMemo(
    () => (isGenerating && generatingFloorId != null ? floors.filter(f => f < generatingFloorId) : floors),
    [floors, isGenerating, generatingFloorId]);
  const navIndex = targetFloorId != null ? availableFloors.indexOf(targetFloorId) : -1;
  const canPrevFloor = navIndex > 0;
  const canNextFloor = navIndex >= 0 && navIndex < availableFloors.length - 1;

  const goNextFloor = useCallback(() => {
    if (!(navIndex >= 0 && navIndex < availableFloors.length - 1)) return;
    if (navIndex + 1 === availableFloors.length - 1) setViewingFloorId(null); // 最后一楼归位
    else setViewingFloorId(availableFloors[navIndex + 1]);
  }, [navIndex, availableFloors]);

  const goPrevFloor = useCallback(() => {
    if (navIndex > 0) setViewingFloorId(availableFloors[navIndex - 1]);
  }, [navIndex, availableFloors]);

  /* ── 发送：/send + /trigger await=true（SKILL.md §S4，坑 14）── */
  const [inputText, setInputText] = useState('');
  const handleSend = useCallback(async () => {
    const trimmed = inputText.trim();
    if (!trimmed || isGenerating) return;
    setInputText('');
    startGenerating();
    try {
      if (!await slash('/send ' + trimmed)) {
        alert('本地预览无酒馆环境，发送不可用（真机验收进酒馆）');
        return;
      }
      await slash('/trigger await=true');   // ★ await 到生成结束，否则提前解锁
    } finally {
      finishGenerating();                    // 无论成败都解锁
    }
  }, [inputText, isGenerating, startGenerating, finishGenerating]);

  /* ── 立绘舞台装配 + 预加载（与阶段二完全相同，照抄 10 章）── */
  const [cast, setCast] = useState<StageCharacter[]>([]);
  const prevLocRef = useRef<string | null>(null);
  useEffect(() => { /* …阶段二装配状态机原样… */ }, [line]);
  useEffect(() => { /* …预加载原样… */ }, [script]);

  const next = useCallback(() => {
    if (!line) return;
    if (isTyping) { skipRef.current = true; return; }
    if (index < script.length - 1) setIndex(i => i + 1);
    else goNextFloor();                       // 本楼到头 → 下一楼
  }, [line, isTyping, index, script.length, goNextFloor]);

  const prev = useCallback(() => {
    if (index > 0) setIndex(i => i - 1);
    else goPrevFloor();                       // 本楼第一张再回退 → 上一楼
  }, [index, goPrevFloor]);

  useEffect(() => { /* 键盘映射与阶段一相同 */ }, [next, prev]);

  /* ── 渲染：阶段二结构 + 楼层条 + 输入栏（JSX 见下）── */
  const bg = sceneOf(line?.location?.path);
  if (!line) {
    return <div className="gal-root"><div className="gal-empty">
      {hasTavern ? '本楼无 <content> 正文' : '本地预览模式：无酒馆数据'}
    </div></div>;
  }
  return (
    <div className="gal-root" onClick={next}>
      {bg && <img key={line.location?.path ?? 'none'} className="gal-bg" src={bg} alt="" />}
      <div className="gal-stage">{/* 阶段二立绘层原样 */}</div>

      {/* 楼层条：位置 + 翻楼按钮（生成中按钮按 availableFloors 自动禁用） */}
      <div className="gal-floorbar" onClick={e => e.stopPropagation()}>
        <button disabled={!canPrevFloor} onClick={goPrevFloor}>‹ 上一楼</button>
        <span>{targetFloorId ?? '—'} 楼{isViewingHistory = (viewingFloorId !== null) ? '（历史）' : ''}</span>
        <button disabled={!canNextFloor} onClick={goNextFloor}>下一楼 ›</button>
        {viewingFloorId !== null && <button onClick={() => setViewingFloorId(null)}>回到最新</button>}
        {isGenerating && <span className="gal-gening">生成中…</span>}
      </div>

      <div className="gal-textbox">
        {line.speaker && (
          <div className={'gal-name' + (line.speaker === '<user>' ? ' gal-name-user' : '')}>
            {line.speaker === '<user>' ? '我' : line.speaker}
          </div>
        )}
        <TypingText key={index} text={line.text} speedMs={SPEED_MS} lineType={line.type}
          onTypingStateChange={setIsTyping} skipRef={skipRef} />
        <div className="gal-progress">{index + 1} / {script.length}</div>
      </div>

      {/* 输入栏：真实发送入口（点击不冒泡到翻页） */}
      <div className="gal-inputbar" onClick={e => e.stopPropagation()}>
        <textarea value={inputText} rows={1} disabled={isGenerating}
          onChange={e => setInputText(e.target.value)}
          onKeyDown={e => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); handleSend(); } }}
          placeholder={isGenerating ? '生成中…' : '输入消息（Enter 发送）'} />
        <button disabled={!inputText.trim() || isGenerating} onClick={handleSend}>发送</button>
      </div>
    </div>
  );
}
```
