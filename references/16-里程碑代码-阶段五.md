## src/fullscreen.ts（全文）

```ts
/* ═══ 伪全屏 + CSS 藏楼 + iframe 高度守卫 ═══
 * 源码：GameScreen.tsx:255-336（伪全屏）+ iframeGuard.ts 全文（守卫）。 */

const STYLE_ID = 'galmilestone-fs-hide';
declare global { interface Window { __galmilestoneFullscreen?: boolean; } }

/** 父页 jQuery（同源前提；拿不到 → 所有父页操作静默放弃） */
export function getParentJQuery(): any | null {
  try {
    if (window.parent && window.parent !== window) {
      const p$ = (window.parent as any).$;
      if (p$) return p$;
    }
  } catch { /* 跨域 */ }
  return null;
}

/** 自身 iframe 元素 */
export function getSelfIframe(): HTMLIFrameElement | null {
  try { return window.frameElement as HTMLIFrameElement | null; } catch { return null; }
}

/** 目标高度：视口比默认矮就适配视口 */
function computeTargetHeight(isMobile: boolean): number {
  const defaultH = isMobile ? 700 : 800;
  try {
    const parentH = window.parent.innerHeight;
    if (parentH > 0 && parentH < defaultH + 40) return Math.max(400, parentH - 20);
  } catch { /* 跨域 */ }
  return defaultH;
}

/** 自己所在楼层容器（.mes[mesid]），父页查询失败回退取最后一楼 */
function closestMes(p$: any) {
  const iframe = getSelfIframe();
  return iframe ? p$(iframe).closest('.mes') : p$('#chat .mes').last();
}

/* ── 进入伪全屏：CSS 藏楼 + .mes 顶满视口 + 原生全屏尽力而为 ── */
export async function enterFullscreen(): Promise<boolean> {
  const p$ = getParentJQuery();
  if (!p$) return false;
  try {
    const $mes = closestMes(p$);
    let hide = p$(`#${STYLE_ID}`);
    if (hide.length === 0) hide = p$(`<style id="${STYLE_ID}"></style>`).appendTo('head');
    const floorId = $mes.attr('mesid');
    hide.text(floorId
      ? `#chat .mes:not([mesid="${floorId}"]) { display: none !important; }`
      : `#chat .mes { display: none !important; }`);
    $mes.css({ position: 'fixed', top: 0, left: 0, width: '100vw', height: '100vh',
      'z-index': 99999, 'max-width': 'none', 'max-height': 'none' });
    const iframe = getSelfIframe();
    if (iframe) p$(iframe).css({ width: '100%', height: '100%' });
    (window as any).__galmilestoneFullscreen = true;
    try { await document.documentElement.requestFullscreen(); } catch { /* 伪全屏兜底（坑 25）*/ }
    return true;
  } catch (e) { console.warn('[gal] 进入全屏失败', e); return false; }
}

/* ── 退出：清样式 + 摘注入 + 复位标志 + 重撑高度 ── */
export async function exitFullscreen(restoreH?: () => void): Promise<void> {
  const p$ = getParentJQuery();
  if (!p$) return;
  try {
    if (document.fullscreenElement) { try { await document.exitFullscreen(); } catch { /* noop */ } }
    const $mes = closestMes(p$);
    $mes.css({ position: '', top: '', left: '', width: '', height: '',
      'z-index': '', 'max-width': '', 'max-height': '' });
    p$(`#${STYLE_ID}`).remove();
    (window as any).__galmilestoneFullscreen = false;
    restoreH?.();
  } catch (e) { console.warn('[gal] 退出全屏失败', e); }
}

/* ── iframe 高度守卫（事件驱动，禁暴力轮询；SKILL.md §S6 高度守卫）── */
export interface GuardHandle { force(): void; burst(): void; destroy(): void; }

export function startHeightGuard(isMobile: boolean): GuardHandle {
  const p$ = getParentJQuery();
  const iframe = getSelfIframe();
  if (!p$ || !iframe) return { force() {}, burst() {}, destroy() {} };

  let rafId = 0; let destroyed = false;
  const applyHeight = () => {
    if (destroyed || (window as any).__galmilestoneFullscreen) return;   // 全屏时跳过
    try {
      const targetH = computeTargetHeight(isMobile);
      const cur = p$(iframe).height();
      if (Math.abs(cur - targetH) > 1) p$(iframe).css({ height: `${targetH}px` }); // 差值>1px 才写
    } catch { /* noop */ }
  };
  const schedule = () => { if (rafId || destroyed) return; rafId = requestAnimationFrame(() => { rafId = 0; applyHeight(); }); };

  applyHeight();
  const observers: MutationObserver[] = [];
  try { const o = new MutationObserver(schedule); o.observe(iframe, { attributes: true, attributeFilter: ['style'] }); observers.push(o); } catch { /* noop */ }
  try {
    const parentEl = iframe.parentElement;
    if (parentEl) { const o = new MutationObserver(schedule); o.observe(parentEl, { attributes: true, attributeFilter: ['style', 'class'] }); observers.push(o); }
  } catch { /* noop */ }
  try { new ResizeObserver(schedule).observe(document.body); } catch { /* noop */ }
  const onResize = () => schedule();
  window.addEventListener('resize', onResize);

  return {
    force: applyHeight,
    burst: () => { applyHeight(); setTimeout(applyHeight, 100); setTimeout(applyHeight, 200); }, // 退出全屏修复竞态
    destroy: () => {
      destroyed = true;
      if (rafId) cancelAnimationFrame(rafId);
      observers.forEach(o => o.disconnect());
      window.removeEventListener('resize', onResize);
      try { p$(iframe).css({ height: '' }); } catch { /* noop */ }
    },
  };
}
```

## App.tsx 追加（三处插入）

```tsx
// ① import 区：
import { enterFullscreen, exitFullscreen, startHeightGuard, type GuardHandle } from './fullscreen';

// ② 组件内：挂守卫 + 监听原生全屏退出（用户按 Esc）→ burst 修复 + 同步退伪全屏
const [isFs, setIsFs] = useState(false);
const guardRef = useRef<GuardHandle | null>(null);
useEffect(() => {
  guardRef.current = startHeightGuard(false);            // 手机端传 true
  const onFsChange = async () => {
    if (!document.fullscreenElement && isFs) {
      await exitFullscreen(() => guardRef.current?.burst());
      setIsFs(false);
    }
  };
  document.addEventListener('fullscreenchange', onFsChange);
  return () => { document.removeEventListener('fullscreenchange', onFsChange); guardRef.current?.destroy(); };
}, [isFs]);

// ③ 楼层条里追加全屏按钮：
<button onClick={async () => {
  if (isFs) { await exitFullscreen(() => guardRef.current?.burst()); setIsFs(false); }
  else if (await enterFullscreen()) setIsFs(true);
}}>{isFs ? '退出全屏' : '全屏'}</button>
```

## lock-frontend.ts（独立脚本全文——不进前端构建，单独发布）

```ts
/**
 * <项目名>-锁定前端 —— 独立酒馆助手脚本
 * 策略：向父页注入 <script>，在父页 JS 引擎里包掉三条删除路径，
 * 凡指向锁定楼 iframe 的一律静默拒删；另以 CSS 藏掉其余楼层（含锁定期新楼）。
 * 触发：父页 fullscreenchange——全屏谁就锁谁，退出全屏解锁。
 */
$(() => {
  const STYLE_ID = 'galmilestone-lock-floor-style';
  const PROTECT_ID = 'galmilestone-lock-protect-script';
  const parentDoc = window.parent.document;
  const parentWin = window.parent;

  let lockedFloorId: number | null = null;
  let hideObserver: MutationObserver | null = null;

  /* ── CSS 藏楼 ── */
  function updateHideStyle(keepFloorId: number | null) {
    parentDoc.getElementById(STYLE_ID)?.remove();
    if (keepFloorId === null) return;
    const style = parentDoc.createElement('style');
    style.id = STYLE_ID;
    style.textContent = `.mes:not([mesid="${keepFloorId}"]) { display: none !important; }`;
    parentDoc.head.appendChild(style);
  }

  /* ── 三条删除路径拦截（注入父页执行）── */
  function injectProtection(floorId: number) {
    removeProtection();
    const script = parentDoc.createElement('script');
    script.id = PROTECT_ID;
    script.textContent = `
      (function() {
        if (window.__galmilestoneLockCleanup) window.__galmilestoneLockCleanup();
        var LOCKED = ${floorId};
        var _iframeRemove = HTMLIFrameElement.prototype.remove;
        HTMLIFrameElement.prototype.remove = function() {
          var mes = this.closest('[mesid]');
          if (mes && parseInt(mes.getAttribute('mesid'), 10) === LOCKED) return;   // ① 原生 remove
          return _iframeRemove.call(this);
        };
        var _removeChild = Node.prototype.removeChild;
        Node.prototype.removeChild = function(child) {
          if (child instanceof HTMLIFrameElement) {
            var mes = child.closest('[mesid]');
            if (mes && parseInt(mes.getAttribute('mesid'), 10) === LOCKED) return child;
          }
          if (child && child.querySelectorAll &&
              child.querySelectorAll('.mes[mesid="' + LOCKED + '"] iframe').length > 0) return child; // ② removeChild 含嵌套
          return _removeChild.call(this, child);
        };
        if (window.$ && window.$.fn) {
          var _jqRemove = window.$.fn.remove;
          window.$.fn.remove = function() {                                          // ③ jQuery remove
            var self = this;
            for (var i = 0; i < self.length; i++) {
              var el = self[i];
              if (el instanceof HTMLIFrameElement) {
                var mes = el.closest('[mesid]');
                if (mes && parseInt(mes.getAttribute('mesid'), 10) === LOCKED) { self.splice(i, 1); i--; }
              }
            }
            return self.length > 0 ? _jqRemove.call(self) : self;
          };
        }
        window.__galmilestoneLockCleanup = function() {
          HTMLIFrameElement.prototype.remove = _iframeRemove;
          Node.prototype.removeChild = _removeChild;
          if (window.$ && window.$.fn && _jqRemove) window.$.fn.remove = _jqRemove;
          delete window.__galmilestoneLockCleanup;
        };
      })();
    `;
    parentDoc.head.appendChild(script);
  }

  function removeProtection() {
    try { (parentWin as any).__galmilestoneLockCleanup?.(); } catch { /* noop */ }
    parentDoc.getElementById(PROTECT_ID)?.remove();
  }

  function lockFloor(floorId: number) {
    if (lockedFloorId === floorId) return;
    if (lockedFloorId !== null) unlockFloor();
    lockedFloorId = floorId;
    updateHideStyle(floorId);
    injectProtection(floorId);
  }
  function unlockFloor() {
    if (lockedFloorId === null) return;
    removeProtection();
    updateHideStyle(null);
    lockedFloorId = null;
  }

  /* ── 触发：父页 fullscreenchange（脚本后加载时若已全屏也补一次）── */
  function floorIdOf(el: Element): number | null {
    const mes = el.closest('[mesid]');
    if (!mes) return null;
    const id = parseInt(mes.getAttribute('mesid')!, 10);
    return isNaN(id) ? null : id;
  }
  $(parentDoc).on('fullscreenchange', () => {
    const fsEl = parentDoc.fullscreenElement;
    if (!fsEl) { unlockFloor(); return; }
    const id = floorIdOf(fsEl);
    if (id !== null) lockFloor(id);
  });
  const cur = parentDoc.fullscreenElement;
  if (cur) { const id = floorIdOf(cur); if (id !== null) lockFloor(id); }

  /* ── 锁定期新楼当场藏掉 ── */
  hideObserver = new MutationObserver(muts => {
    if (lockedFloorId === null) return;
    for (const m of muts) for (const node of Array.from(m.addedNodes)) {
      if (node instanceof HTMLElement && node.matches(`.mes:not([mesid="${lockedFloorId}"])`)) {
        node.style.display = 'none';
      }
    }
  });
  hideObserver.observe(parentDoc.getElementById('chat') || parentDoc.body, { childList: true, subtree: true });

  $(window).on('pagehide', () => { unlockFloor(); hideObserver?.disconnect(); });
});
```

## 自测清单（**真机门**——全部在真实酒馆里验）

1. 点"全屏"：父页 head 出现 `galmilestone-fs-hide` 样式；画面里除游戏外无任何楼层；原生全屏不可用时不影响游玩。
2. 按 Esc：样式摘除、楼层恢复、iframe 高度正确（无塌缩/双滚动条——burst 生效）。
3. 全屏期间发消息生成新楼：新楼到达但不可见，画面纹丝不动。
4. 启用锁定前端后再全屏：父页 head 出现 `-lock-protect-script`；从酒馆 UI 删除锁定楼 → 被静默拒绝，console 出现不了删除后的黑屏，游戏继续。
5. 卸载/关闭锁定前端脚本重复第 4 步 → 楼层被删（对照实验，确认拦截真的在工作）。
6. 卡住了 → `11-排障决策树.md` 现象 14/15/16。

至此六个阶段的里程碑齐了：09/10（本地可跑）→ 14（真机·发送与楼层）→ 15（真机·删楼重roll）→ 16（真机·守卫）。阶段六组件按 `06` 章逐个加装，每装一个跑一遍该节验收门。
