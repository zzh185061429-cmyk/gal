## src/tavernOps.ts（全文）

```ts
import { getMessages, lastMessageId, slash, emit, chatVars, STORY_UPDATED } from './tavern';
import { stripThinking } from './parser';

const g = globalThis as any;
const mvu = (): any => g.Mvu ?? null;

/* ── 读变量：楼层 stat_data → 聊天变量 → null（SKILL.md §S5 降级链）── */
export function readStatData(messageId?: number): Record<string, any> | null {
  const m = mvu();
  if (m) {
    try {
      const sd = m.getMvuData({ type: 'message', message_id: messageId ?? 'latest' })?.stat_data;
      if (sd && typeof sd === 'object') return sd;
    } catch { /* 楼层读取失败 → 回退聊天变量 */ }
  }
  if (messageId == null) {
    const sd = chatVars()?.stat_data;
    if (sd && typeof sd === 'object') return sd;
  }
  return null;
}

/* ── 删楼：/cut 彻底删（deleteChatMessages 只清空内容——坑 19）+ 事件刷新（坑 24）── */
export async function deleteFloors(start: number, end: number): Promise<boolean> {
  if (start < 0 || end < start) return false;
  const ok = await slash(`/cut ${start}-${end}`);
  if (ok) emit(STORY_UPDATED);            // 前端自刷新，绝不 reload
  return ok;
}

/* ── 静默重roll：六步全流程（SKILL.md §S5 重roll 七步）── */
const REGEN_ASSISTANT_FLOORS = 5;

export async function regenerateCurrentFloor(): Promise<{ success: boolean; error?: string }> {
  try {
    if (typeof g.generate !== 'function' || typeof g.setChatMessages !== 'function') {
      return { success: false, error: '需要酒馆环境' };
    }

    /* ① 最后一楼必须 AI 楼 */
    const lastFloorId = lastMessageId();
    const lastFloor = getMessages(-1)[0];
    if (!lastFloor) return { success: false, error: '未找到最后一楼层' };
    if (lastFloor.role !== 'assistant') return { success: false, error: '最后一楼层不是 assistant' };

    /* ② 向前找最近的 user 楼作为输入 */
    let userText = ''; let userFloorId = -1;
    for (let i = (lastFloorId as number) - 1; i >= 0; i--) {
      const m = getMessages(i)[0];
      if (m?.role === 'user') { userText = m.message || ''; userFloorId = i; break; }
    }
    if (!userText) return { success: false, error: '未找到上一层的用户输入' };

    /* ③ 历史窗口只留最近 5 层 AI 楼——chat_history override 会绕过 /hide，不裁就裸发全部历史（坑 20） */
    const all = getMessages('0-' + userFloorId);
    let windowed = all; let seen = 0;
    for (let i = all.length - 1; i >= 0; i--) {
      if (all[i].role === 'assistant' && ++seen >= REGEN_ASSISTANT_FLOORS) { windowed = all.slice(i); break; }
    }
    const historyPrompts = windowed.map((m: any) => ({ role: m.role, content: m.message || '' }));

    /* ④ 静默生成：只生成，不创建新楼层 */
    const raw = await g.generate({
      user_input: userText,
      should_stream: false,
      should_silence: true,
      overrides: { chat_history: { prompts: historyPrompts } },
    });
    if (!raw || typeof raw !== 'string') return { success: false, error: 'AI 返回空响应' };

    /* ⑤ 剥思维链 + 提取主体（<maintext> 或全文，保住 <content> 结构原样回写）*/
    const filtered = stripThinking(raw);
    const mt = filtered.match(/<maintext>([\s\S]*?)<\/maintext>/i);
    const maintext = (mt ? mt[1] : filtered).trim();

    /* ⑤' 变量也要跟着重算：以该楼旧变量为基线解析新回复（坑 21）。
     * 注意吃的是剥链"全文"——<UpdateVariable> 块在 <content> 外面（SKILL.md §S5）。 */
    try {
      if (typeof g.waitGlobalInitialized === 'function') await g.waitGlobalInitialized('Mvu');
      const m = mvu();
      if (m) {
        const oldData = m.getMvuData({ type: 'message', message_id: lastFloorId });
        await m.parseMessage(filtered, oldData);
      }
    } catch { console.warn('[gal][regen] 变量解析失败，沿用当前 MVU 数据'); }

    /* ⑥ 原位替换（不删不建，楼号不变）+ 事件刷新（不重载不退全屏） */
    await g.setChatMessages([{ message_id: lastFloorId, message: maintext }], { refresh: 'none' });
    emit(STORY_UPDATED);
    return { success: true };
  } catch (err: any) {
    return { success: false, error: err?.message || '重新生成失败' };
  }
}
```

## src/VariablesLite.tsx（全文）

```tsx
import React, { useState } from 'react';
import { readStatData } from './tavernOps';

/** 极简变量查看器：调阶段三/四时全靠它看 stat_data */
export function VariablesLite({ floorId }: { floorId: number | null }) {
  const [open, setOpen] = useState(false);
  const sd = open ? readStatData(floorId ?? undefined) : null;
  return (
    <div className="gal-varslite" onClick={e => e.stopPropagation()}>
      <button onClick={() => setOpen(o => !o)}>{open ? '收起变量' : '变量'}</button>
      {open && (
        <pre>{sd ? JSON.stringify(sd, null, 2) : '（无 stat_data：未装 MVU 框架或空聊天——属正常降级）'}</pre>
      )}
    </div>
  );
}
```

## 自测清单（**真机门**）

1. 变量面板：装了 MVU 的聊天能读出 `stat_data`；没装的聊天显示降级文案且其它功能不受影响。
2. 重roll：点"重roll 本楼"→ 楼层号不变、正文原位换新、`stat_data` 里被新剧情影响的字段同步变化（时间/地点最直观）。
3. 窗口裁剪：重roll 时 console 看生成请求，历史只带最近 5 层 AI 楼（坑 20）。
4. 删楼：删中间一楼 → 楼层真没了、前端无重载全屏不退、楼号收敛正确。
